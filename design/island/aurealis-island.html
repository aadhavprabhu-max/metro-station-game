The live sim runs a train around the S3 ring. It pops “arriving” alerts, splits off a doors bubble at each station, and you can grab control any time. Every surface is true black #000000. The island morphs frame by frame on springs, and the doors bubble pinches off with a liquid bridge.

Game integration

// To add it to the game, copy the three marked blocks from this file:
//   ISLAND CSS · ISLAND HTML · ISLAND SCRIPT   (it doesn't need anything else on the page)
// Put the HTML inside a position:relative container, then drive it from the game loop:

AurealisIsland.update({
  line: 'S3', color: '#F2B33D', dest: 'Ringbahn',
  prev: 'Altstadt', next: 'Central', after: 'Domplatz',
  etaSec: 95, progress: 0.4,          // 0..1 between prev and next
  platform: '4', speed: 72, delayMin: 0, doors: 'right'
});

AurealisIsland.setState('compact');   // idle | compact | split | expanded | actions
AurealisIsland.doors('closing');      // 'open' | 'closing' | null  → split bubble
AurealisIsland.state;                 // current state (read-only)

AurealisIsland.hide();                // tuck it away (also: show(), toggle(), .hidden)
AurealisIsland.onVisibility = (hidden) => { /* show your bottom glass HUD while hidden */ };
AurealisIsland.alertsWhenHidden = false; // true = alerts pop the island back up
AurealisIsland.alert({ tone: 'warn', icon: 'warn',
  title: 'Delay +2 min', sub: 'Signal check near Domplatz', duration: 3500 });

AurealisIsland.onAction = (name) => { /* 'doors' | 'horn' | 'announce' | 'map' */ };
AurealisIsland.onState  = (state) => { /* fires on every state change */ };

// Esc only closes the island while it's open, and then stops the key there (so the
// game's pause menu doesn't also open). Turn either behaviour off if you want:
AurealisIsland.escapeKey = true;
AurealisIsland.closeOnOutsideClick = true;
